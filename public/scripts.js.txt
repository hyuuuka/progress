// public/script.js
let isLoggedIn = false;

function getStatus(progress) {
    if (progress === 0) return 'Not Done';
    if (progress === 100) return 'Done';
    return 'Partially Done';
}

function getStatusClass(progress) {
    if (progress === 0) return 'status-not-done';
    if (progress === 100) return 'status-done';
    return 'status-partial';
}

async function fetchTasks() {
    const response = await fetch('/api/tasks');
    return await response.json();
}

async function updateTask(index, newProgress) {
    const tasks = await fetchTasks();
    tasks[index].progress = parseInt(newProgress);
    
    await fetch(`/api/tasks/${index}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(tasks[index])
    });
    
    renderDashboard();
}

function calculateOverallProgress(tasks) {
    if (!tasks.length) return 0;
    const total = tasks.reduce((acc, task) => acc + task.progress, 0);
    return Math.round(total / tasks.length);
}

async function renderDashboard() {
    const tasks = await fetchTasks();
    const taskList = document.getElementById('taskList');
    taskList.innerHTML = '';
    
    tasks.forEach((task, index) => {
        const status = getStatus(task.progress);
        const statusClass = getStatusClass(task.progress);
        
        const taskElement = document.createElement('div');
        taskElement.className = 'task';
        taskElement.innerHTML = `
            <div class="task-header">
                <span>${task.name}</span>
                <span class="status ${statusClass}">${status} (${task.progress}%)</span>
            </div>
            <div class="progress-bar">
                <div class="progress-fill" style="width: ${task.progress}%"></div>
            </div>
            ${isLoggedIn ? `
                <div class="edit-controls">
                    <input type="range" class="range-input" value="${task.progress}" 
                        min="0" max="100" step="50" 
                        onchange="updateTask(${index}, this.value)">
                    <button onclick="deleteTask(${index})">Delete</button>
                </div>
            ` : ''}
        `;
        taskList.appendChild(taskElement);
    });

    const overallProgress = calculateOverallProgress(tasks);
    document.getElementById('overallProgress').textContent = overallProgress;
    document.getElementById('overallProgressBar').style.width = `${overallProgress}%`;
}

// Login form handling
document.querySelector('.login-form').addEventListener('submit', async function(e) {
    e.preventDefault();
    const username = document.getElementById('username').value;
    const password = document.getElementById('password').value;

    try {
        const response = await fetch('/api/login', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ username, password })
        });

        if (response.ok) {
            isLoggedIn = true;
            document.getElementById('loginForm').classList.add('hidden');
            document.getElementById('adminControls').classList.remove('hidden');
            renderDashboard();
        } else {
            alert('Invalid credentials!');
        }
    } catch (error) {
        alert('Login failed. Please try again.');
    }
});

function logout() {
    isLoggedIn = false;
    document.getElementById('loginForm').classList.remove('hidden');
    document.getElementById('adminControls').classList.add('hidden');
    document.getElementById('username').value = '';
    document.getElementById('password').value = '';
}

async function addTask() {
    const name = document.getElementById('newTaskName').value;
    const progress = parseInt(document.getElementById('newTaskProgress').value);

    if (!name || isNaN(progress) || progress < 0 || progress > 100) {
        alert('Please enter valid task details!');
        return;
    }

    await fetch('/api/tasks', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name, progress })
    });

    document.getElementById('newTaskName').value = '';
    document.getElementById('newTaskProgress').value = '';
    renderDashboard();
}

async function deleteTask(index) {
    if (confirm('Are you sure you want to delete this task?')) {
        await fetch(`/api/tasks/${index}`, { method: 'DELETE' });
        renderDashboard();
    }
}

// Initial render
renderDashboard();
